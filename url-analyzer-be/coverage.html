func NewScrapeHandler(engine scraper.Engine,

        MaxHTTPBatchSize int) *ScrapeHandler {

func (h *ScrapeHandler) Health(c *gin.Context) {
        c.JSON(http.StatusOK, models.APIResponse{

                        "status": "UP",
                },
        })

// validateURL checks for presence, valid structure, and HTTP/HTTPS scheme.
func validateURL(target string) (string, error) {
        if target == "" {
                return "Missing value", errors.New("url field is required")
        }
        parsed, err := url.ParseRequestURI(target)
        if err != nil {
                return "URL parsing error", errors.New("malformed URL format")
        }
        if parsed.Scheme != "http" && parsed.Scheme != "https" {
                return "Unsupported protocol", errors.New("URL scheme must be http or https")
        }
        if parsed.Host == "" {
                return "Missing hostname", errors.New("URL missing host")
        }
        return "", nil
}

func (h *ScrapeHandler) Scrape(c *gin.Context) {
        var req models.ScrapeRequest

        if err := c.ShouldBindJSON(&req); err != nil {
                c.JSON(http.StatusBadRequest, models.APIResponse{
                        Status: http.StatusBadRequest,
                        Error:  apperrors.NewAPIError(apperrors.CodeInvalidJSON, apperrors.MsgInvalidJSON, err.Error()),
                })
                return
        }

        if len(req.URLs) == 0 {
                c.JSON(http.StatusBadRequest, models.APIResponse{
                        Status: http.StatusBadRequest,
                        Error:  apperrors.NewAPIError(apperrors.CodeEmptyBatch, apperrors.MsgEmptyBatch),
                })
                return
        }

        if len(req.URLs) > h.MaxHTTPBatchSize {
                c.JSON(http.StatusBadRequest, models.APIResponse{
                        Status: http.StatusBadRequest,
                        Error: apperrors.NewAPIError(
                                apperrors.CodeBatchSizeExceeded,
                                apperrors.MsgBatchSizeExceeded,
                                fmt.Sprintf("Batch size %d exceeds maximum limit of %d URLs", len(req.URLs), h.MaxHTTPBatchSize),
                        ),
                })
                return
        }

        ctx, cancel := context.WithTimeout(c.Request.Context(), h.scrapeTimeout)
        defer cancel()

        results := make([]models.ScrapeItemResult, len(req.URLs))
        var wg sync.WaitGroup

        limitConcurrency := make(chan struct{}, h.MaxHTTPBatchConcurrency)

        for i, target := range req.URLs {
                limitConcurrency <- struct{}{}
                wg.Add(1)

                go func(index int, item models.TargetURL) {
                        defer wg.Done()
                        defer func() { <-limitConcurrency }()

                        if detail, err := validateURL(item.URL); err != nil {
                                results[index] = models.ScrapeItemResult{
                                        URL:   item.URL,
                                        Error: apperrors.NewScrapeError(apperrors.CodeInvalidURL, err.Error(), detail),
                                }
                                return
                        }

                        analysis, err := h.engine.Analyze(ctx, item.URL)
                        if err != nil {
                                errCode := apperrors.CodeScrapeFailed
                                errMsg := "Failed to complete webpage analysis"
                                detail := err.Error()

                                if errors.Is(err, context.DeadlineExceeded) {
                                        errCode = apperrors.CodeTimeout
                                        errMsg = apperrors.MsgTimeout
                                        detail = "Execution exceeded configured timeout duration"
                                }

                                results[index] = models.ScrapeItemResult{
                                        URL:   item.URL,
                                        Error: apperrors.NewScrapeError(errCode, errMsg, detail),
                                }
                                return
                        }

                        results[index] = models.ScrapeItemResult{
                                URL:      item.URL,
                                Analysis: analysis,
                        }
                }(i, target)
        }

        wg.Wait()

        c.JSON(http.StatusOK, models.APIResponse{
                Status: http.StatusOK,
                Data:   results,
        })
}

        analysis, err := s.engine.Analyze(ctx, targetURL)
        if err != nil {
                return nil, err
        }

        if analysis == nil {
                return nil, errors.New("analyzer returned empty result")
        }

        return mapToAPIModel(analysis), nil

func mapToAPIModel(src *analyzer.PageAnalysis) *models.PageAnalysis {
        return &models.PageAnalysis{
                URL:               src.URL,
                HTMLVersion:       src.HTMLVersion,
                Title:             src.Title,
                Headings:          src.Headings,
                InternalLinks:     mapLinks(src.InternalLinks),
                ExternalLinks:     mapLinks(src.ExternalLinks),
                InaccessibleLinks: mapLinks(src.InaccessibleLinks),
                HasLoginForm:      src.HasLoginForm,
        }
}

func mapLinks(srcLinks []analyzer.Link) []models.Link {
        if srcLinks == nil {
                return []models.Link{}
        }

        links := make([]models.Link, len(srcLinks))
        for i, l := range srcLinks {
                links[i] = models.Link{
                        URL: l.URL(),
                }
        }
        return links
}

import (
        "context"
        "fmt"
        "io"
        "net/http"
        "net/url"
        "path"
        "strings"
        "sync"
        "time"
)

const (
        defaultWorkers = 10
        defaultTimeout = 15 * time.Second
        maxBodySize    = 2 * 1024 * 1024
)

type CategorizedLinks struct {
        Internal    []Link
        External    []Link
        Validatable []Link
}

type Analyzer struct {
        client         *http.Client
        scraper        Scraper
        workers        int
        ignoredSchemes map[string]bool
}

func New(opts ...Option) *Analyzer {
        var t *http.Transport
        if dt, ok := http.DefaultTransport.(*http.Transport); ok {
                t = dt.Clone()
        } else {
                t = &http.Transport{}
        }
        t.MaxIdleConns = 100
        t.MaxConnsPerHost = 100

        browserHeaderTransport := &browserHeaderTransport{
                base: t,
        }

        a := &Analyzer{
                workers: defaultWorkers,
                scraper: &DefaultHTMLScraper{},
                client: &http.Client{
                        Timeout:   defaultTimeout,
                        Transport: browserHeaderTransport,
                },
                ignoredSchemes: map[string]bool{
                        "javascript": true,
                        "vbscript":   true,
                        "data":       true,
                        "blob":       true,

                        "mailto": true,
                        "tel":    true,
                        "sms":    true,
                        "callto": true,

                        "android-app": true,
                        "ios-app":     true,

                        "file":   true,
                        "ftp":    true,
                        "intent": true,
                },
        }

        for _, opt := range opts {
                opt(a)
        }

        return a
}

func (a *Analyzer) Analyze(ctx context.Context, targetURL string) (*PageAnalysis, error) {
        if err := ctx.Err(); err != nil {
                return nil, err
        }

        if !strings.HasPrefix(targetURL, "http://") && !strings.HasPrefix(targetURL, "https://") {
                targetURL = "https://" + targetURL
        }
        parsedTarget, err := url.ParseRequestURI(targetURL)
        if err != nil {
                return nil, fmt.Errorf("%w: %v", ErrInvalidURL, err)
        }

        req, err := http.NewRequestWithContext(ctx, http.MethodGet, parsedTarget.String(), nil)
        if err != nil {
                return nil, fmt.Errorf("failed to build request: %w", err)
        }

        resp, err := a.client.Do(req)
        if err != nil {
                if ctx.Err() != nil {
                        return nil, ctx.Err()
                }
                return nil, &HTTPError{
                        StatusCode:  0,
                        Description: "Failed to establish connection to the server.",
                }
        }
        defer resp.Body.Close()

        if resp.StatusCode < 200 || resp.StatusCode >= 400 {
                return nil, &HTTPError{
                        StatusCode:  resp.StatusCode,
                        Description: http.StatusText(resp.StatusCode),
                }
        }

        if err := ctx.Err(); err != nil {
                return nil, err
        }

        limitReader := io.LimitReader(resp.Body, maxBodySize)
        scrapeData, err := a.scraper.Scrape(ScrapeRequest{
                Reader:    limitReader,
                TargetURL: parsedTarget,
        })
        if err != nil {
                return nil, fmt.Errorf("%w: %v", ErrScrape, err)
        }

        _, _ = io.Copy(io.Discard, resp.Body) // drain for conn reuse

        if err := ctx.Err(); err != nil {
                return nil, err
        }

        links, inaccessibleLinks, err := a.processAndValidateLinks(ctx, parsedTarget, scrapeData.RawLinks)
        if err != nil {
                return nil, err
        }

        return &PageAnalysis{
                URL:               parsedTarget.String(),
                HTMLVersion:       scrapeData.HTMLVersion,
                Title:             scrapeData.Title,
                Headings:          scrapeData.Headings,
                InternalLinks:     links.Internal,
                ExternalLinks:     links.External,
                InaccessibleLinks: inaccessibleLinks,
                HasLoginForm:      scrapeData.HasLoginForm,
        }, nil
}

func (a *Analyzer) processAndValidateLinks(ctx context.Context, baseURL *url.URL, rawLinks []Link) (CategorizedLinks, []Link, error) {
        links := a.categorizeLinks(baseURL, rawLinks)

        inaccessibleLinks, err := a.getInaccessibleLinks(ctx, links.Validatable)
        if err != nil {
                return CategorizedLinks{}, nil, err
        }

        return links, inaccessibleLinks, nil
}

func (a *Analyzer) categorizeLinks(baseURL *url.URL, rawLinks []Link) CategorizedLinks {

        categorized := CategorizedLinks{
                Validatable: make([]Link, 0, len(rawLinks)),
                Internal:    make([]Link, 0, len(rawLinks)/2),
                External:    make([]Link, 0, len(rawLinks)/2),
        }

        seen := make(map[string]struct{}, len(rawLinks))

        for _, rawLink := range rawLinks {
                rawURLStr := strings.TrimSpace(rawLink.URL())
                if rawURLStr == "" {
                        continue
                }

                resolvedURL, err := baseURL.Parse(rawURLStr)
                if err != nil {
                        continue
                }

                scheme := strings.ToLower(resolvedURL.Scheme)
                if a.ignoredSchemes[scheme] {
                        continue
                }

                normalizedURL := a.NormalizeURL(resolvedURL)
                normalizedStr := normalizedURL.String()

                if _, exists := seen[normalizedStr]; exists {
                        continue
                }
                seen[normalizedStr] = struct{}{}

                scrapedLink := ScrapedLink(normalizedStr)
                categorized.Validatable = append(categorized.Validatable, scrapedLink)

                if normalizedURL.Host == baseURL.Host || normalizedURL.Host == "" {
                        categorized.Internal = append(categorized.Internal, scrapedLink)
                } else {
                        categorized.External = append(categorized.External, scrapedLink)
                }
        }

        return categorized
}

func (a *Analyzer) NormalizeURL(u *url.URL) *url.URL {
        cloned := *u

        cloned.Scheme = strings.ToLower(cloned.Scheme)
        cloned.Host = strings.ToLower(cloned.Host)

        port := cloned.Port()
        if (cloned.Scheme == "http" && port == "80") ||
                (cloned.Scheme == "https" && port == "443") {
                cloned.Host = cloned.Hostname()
        }

        cloned.Fragment = ""
        cloned.RawFragment = ""

        if cloned.RawQuery != "" {
                cloned.RawQuery = cloned.Query().Encode()
        }

        if cloned.Path != "" {

                cloned.Path = path.Clean(cloned.Path)

                if cloned.Path == "/" {
                        cloned.Path = ""
                }
        }

        return &cloned
}

func (a *Analyzer) getInaccessibleLinks(ctx context.Context, links []Link) ([]Link, error) {
        if len(links) == 0 {
                return nil, nil
        }

        jobs := make(chan Link, len(links))
        results := make(chan Link, len(links))
        var wg sync.WaitGroup

        numWorkers := a.workers
        if len(links) < numWorkers {
                numWorkers = len(links)
        }

        // Start Workers
        for i := 0; i < numWorkers; i++ {
                wg.Add(1)
                go func() {
                        defer wg.Done()

                        for link := range jobs {

                                if ctx.Err() != nil {
                                        return
                                }

                                if a.isLinkInaccessible(ctx, link) {
                                        select {
                                        case <-ctx.Done():
                                                return
                                        case results <- link:
                                        }
                                }
                        }
                }()
        }

        for _, link := range links {
                jobs <- link
        }
        close(jobs)

        go func() {
                wg.Wait()
                close(results)
        }()

        var inaccessible []Link
        for brokenLink := range results {
                inaccessible = append(inaccessible, brokenLink)
        }

        if err := ctx.Err(); err != nil {
                return nil, err
        }

        return inaccessible, nil
}

func (a *Analyzer) isLinkInaccessible(ctx context.Context, link Link) bool {

        inaccessible, err := a.checkURL(ctx, http.MethodHead, link.URL())
        if err == nil && !inaccessible {
                return false
        }

        inaccessible, err = a.checkURL(ctx, http.MethodGet, link.URL())
        if err != nil {
                return true
        }

        return inaccessible
}

func (a *Analyzer) checkURL(ctx context.Context, method, targetURL string) (bool, error) {
        req, err := http.NewRequestWithContext(ctx, method, targetURL, nil)
        if err != nil {
                return true, err
        }

        resp, err := a.client.Do(req)
        if err != nil {
                return true, err
        }
        defer resp.Body.Close()

        return resp.StatusCode >= 400, nil // ignoring 301,302 redirects for now
}

        base http.RoundTripper
}

func (t *browserHeaderTransport) RoundTrip(req *http.Request) (*http.Response, error) {
        clonedReq := req.Clone(req.Context())

        clonedReq.Header.Set("User-Agent", browserUserAgent)
        clonedReq.Header.Set("Accept", browserAccept)
        clonedReq.Header.Set("Accept-Language", browserLanguage)

        return t.base.RoundTrip(clonedReq)
}

import (
        "io"
        "net/url"
        "strings"

        "golang.org/x/net/html"
)

// Link represents any link or resource URL extracted from a parsed document.
//
// We use an interface so the analyzer can handle different link types dynamically.
// Higher-level analyzer components can perform type assertions (e.g., link.(FormLink))
// to run type-specific logic, or inspect validation status.
//
// Examples of future link implementations:
//
//   - AnchorLink: standard text links (<a href="...">)
//
//   - MediaLink: images, scripts, or videos (<img src="...">, <script src="...">)
//
//   - FormLink: form action targets (<form action="..." method="POST">)
//
//   - AnalyzedLink: holds validation status and potential accessibility errors:
//
//     type AnalyzedLink struct {
//     Link               // embeds the inner Link interface
//     ValidationError error // e.g., HTTP 404, DNS resolution failure
type Link interface {
        URL() string
}

type ScrapedLink string

func (l ScrapedLink) URL() string {
        return string(l)
}

type ScrapeRequest struct {
        Reader    io.Reader
        TargetURL *url.URL
}

type Scraper interface {
        Scrape(req ScrapeRequest) (*ScrapeData, error)
}

type DefaultHTMLScraper struct{}

func (s *DefaultHTMLScraper) Scrape(req ScrapeRequest) (*ScrapeData, error) {
        doc, err := html.Parse(req.Reader)
        if err != nil {
                return nil, err
        }

        data := &ScrapeData{
                HTMLVersion: "Unknown",
                Headings:    make(map[string]int),
                RawLinks:    make([]Link, 0),
        }

        data.HTMLVersion = extractDoctype(doc)
        data.Title = extractHeadTitle(doc)

        s.walk(doc, data)

        return data, nil
}

func (s *DefaultHTMLScraper) walk(n *html.Node, data *ScrapeData) {
        if n.Type == html.ElementNode {
                tagName := strings.ToLower(n.Data)

                // 1. Record Headings (h1..h6)
                if len(tagName) == 2 && tagName[0] == 'h' && tagName[1] >= '1' && tagName[1] <= '6' {
                        data.Headings[tagName]++
                }

                // 2. Extract, clean, and resolve links
                if linkURL := extractLinkFromNode(n, tagName); linkURL != "" {
                        data.RawLinks = append(data.RawLinks, ScrapedLink(linkURL))
                }

                // 3. Check for Password Input inside <form>
                if tagName == "form" && !data.HasLoginForm {
                        data.HasLoginForm = hasPasswordInput(n)
                }
        }

        for c := n.FirstChild; c != nil; c = c.NextSibling {
                s.walk(c, data)
        }
}

// Helper to target key attributes (href, src, action, poster) per HTML specification
func extractLinkFromNode(n *html.Node, tagName string) string {
        var targetAttr string

        switch tagName {
        case "a", "link", "area", "base":
                targetAttr = "href"
        case "script", "img", "iframe", "embed", "source", "track":
                targetAttr = "src"
        case "form":
                targetAttr = "action"
        case "video":
                targetAttr = "poster"
        default:
                return ""
        }

        for _, attr := range n.Attr {
                if strings.ToLower(attr.Key) == targetAttr {
                        if attr.Val != "" {
                                return normalizeURL(attr.Val)
                        }
                }
        }

        return ""
}

func normalizeURL(rawURL string) string {
        parsed, err := url.Parse(rawURL)
        if err != nil {
                return rawURL
        }

        return parsed.String()
}

func extractHeadTitle(n *html.Node) string {
        headNode := findElement(n, "head")
        if headNode == nil {
                return ""
        }

        titleNode := findElement(headNode, "title")
        if titleNode == nil {
                return ""
        }

        return strings.TrimSpace(extractTextContent(titleNode))
}

func extractTextContent(n *html.Node) string {
        if n.Type == html.TextNode {
                return n.Data
        }
        var sb strings.Builder
        for c := n.FirstChild; c != nil; c = c.NextSibling {
                sb.WriteString(extractTextContent(c))
        }
        return sb.String()
}

func findElement(n *html.Node, targetTag string) *html.Node {
        if n.Type == html.ElementNode && strings.ToLower(n.Data) == targetTag {
                return n
        }
        for c := n.FirstChild; c != nil; c = c.NextSibling {
                if res := findElement(c, targetTag); res != nil {
                        return res
                }
        }
        return nil
}

func hasPasswordInput(formNode *html.Node) bool {
        if formNode.Type == html.ElementNode && strings.ToLower(formNode.Data) == "input" {
                for _, attr := range formNode.Attr {
                        if strings.ToLower(attr.Key) == "type" && strings.ToLower(attr.Val) == "password" {
                                return true
                        }
                }
        }
        for c := formNode.FirstChild; c != nil; c = c.NextSibling {
                if hasPasswordInput(c) {
                        return true
                }
        }
        return false
}

func extractDoctype(doc *html.Node) string {
        for c := doc.FirstChild; c != nil; c = c.NextSibling {
                if c.Type == html.DoctypeNode {
                        return parseDoctype(c)
                }
        }
        return "HTML (Legacy/Unknown)"
}

func parseDoctype(node *html.Node) string {
        var sb strings.Builder
        sb.WriteString(node.Data)

        for _, attr := range node.Attr {
                sb.WriteString(" ")
                sb.WriteString(attr.Val)
        }

        dt := strings.ToLower(sb.String())

        if strings.Contains(dt, "html 4.01") {
                return "HTML 4.01"
        }
        if strings.Contains(dt, "xhtml 1.0") {
                return "XHTML 1.0"
        }
        if strings.TrimSpace(strings.ToLower(node.Data)) == "html" {
                return "HTML5"
        }

        return "HTML (Legacy/Unknown)"
}
